\block{Background}{
Anyone investing money faces one decision: how much to put in each asset. Portfolio optimisation answers it by estimating each asset's expected return and risk from past prices and solving for the best mix. In practice the estimates are noisy, and real investors add rules the textbook ignores: a limit on the number of holdings, a cash floor, sector caps, trading costs.

\statement{Real-world investing imposes constraints on portfolio optimisation. We ask how the choice of risk model changes the outcome once those constraints are fixed, and whether any model earns its complexity over the simplest alternative, equal weight.}

\textbf{What we ran.} 5 investor rule-sets $\times$ 5 risk models (Markowitz, Ledoit--Wolf, robust, CVaR, equal weight), walk-forward over 2016--2025 with trading costs.

\textbf{Limitations.} One decade, one market (US large caps), no short selling or leverage, and a universe chosen with hindsight: all 40 stocks still trade today.
}

\block{1. Experimental Setup}{
\begin{itemize}
\item \textbf{Universe:} 43 assets: 40 US large caps (four per GICS sector), a bond ETF, gold, cash.
\item \textbf{Prices:} daily, 2008--2025. Tune on 2011--2015, test on 2016--2025.
\item \textbf{Rule-sets:} five archetypes, values derived from Vanguard allocations and the 2022 Survey of Consumer Finances.
\end{itemize}

\block{2. The Portfolio Problem}{
{\renewcommand{\arraystretch}{1.35}
\begin{tabular}{@{}l@{\hspace{1.2em}}l@{}}
$\w\in\R^{n}$ & weights: fraction of the portfolio in each of $n$ assets \\
$z_a\in\{0,1\}$ & whether asset $a$ is held \\
$b_a,\, s_a \ge 0$ & amount bought and sold, relative to the current weights $\w^{\text{prev}}$ \\
$\hat\mub,\; \hat\Sig,\; r_s$ & expected daily return, covariance, and day-$s$ return vector, estimated from the last $|S|$ days \\
$h = 21$, \; $\tau = 10$\,bp & expected holding period (days); cost per unit traded \\
$\rho_{\max}(t)$ & risk limit, tightening linearly as the horizon shrinks \\
\end{tabular}}

\vspace{0.4em}
{\Large
\[
\max_{\w,\,z,\,b,\,s}\;\; h\,\hat\mub^{\top}\w \;-\; \tau\sum_a (b_a + s_a)
\qquad\text{s.t.}\qquad \rho(\w)\;\le\;\rho_{\max}(t)
\]}
\vspace{-0.2em}
{\Large\renewcommand{\arraystretch}{1.6}
\begin{tabular}{@{}l@{\hspace{2.5em}}l@{}}
$\textstyle\sum_a w_a = 1$ & \large budget \\
$w_a = w^{\text{prev}}_a + b_a - s_a$ & \large trade balance \\
$w^{\min} z_a \;\le\; w_a \;\le\; w^{\max}_a z_a$ & \large minimum and maximum position \\
$\textstyle\sum_{a\,\notin\,\text{cash}} z_a \;\le\; K$ & \large cardinality: at most $K$ names \\
$\textstyle\sum_{a\in\text{sector }k} w_a \;\le\; c_k$ & \large sector caps \\
$w_{\text{cash}} \;\ge\; c_{\min}$ & \large cash floor \\
\end{tabular}}
}

\block{3. Five Risk Models}{
{\Large
\begin{align*}
\model{Markowitz} \qquad & \rho(\w) = \w^{\top}\hat\Sig\,\w \\[0.3em]
\model{Markowitz + Ledoit--Wolf} \qquad & \rho(\w) = \w^{\top}\hat\Sig_{\text{LW}}\,\w, \qquad \hat\Sig_{\text{LW}} = (1-\delta)\hat\Sig + \delta\,\bar\sigma^{2} I \\[0.3em]
\model{Robust} \qquad & \rho(\w) = \w^{\top}\hat\Sig\,\w, \qquad \hat\mub^{\top}\w \;\to\; \hat\mub^{\top}\w - \kappa\sqrt{\w^{\top}\Omega\,\w}, \quad \Omega = \hat\Sig/|S| \\[0.3em]
\model{CVaR} \qquad & \rho(\w) = \zeta + \frac{1}{(1-\alpha)|S|}\sum_{s\in S} u_s, \qquad \alpha = 0.95 \\
 & u_s \;\ge\; -\,r_s^{\top}\w - \zeta, \qquad u_s \ge 0 \\[0.3em]
\model{Equal weight} \qquad & w_a = 1/N
\end{align*}}
}

\block{4. When the Portfolio Is Reweighted}{
Starting from the weights currently held, $\w^{\text{prev}}$, the portfolio is re-solved when any of the following holds ($\w^{\text{target}}$ = weights chosen at the last solve; $\sigma_{21},\sigma_{252}$ = market volatility over the last 21 and 252 days):
{\Large
\begin{align*}
&\text{calendar} && 63 \text{ trading days since the last solve} \\
&\text{drift} && \textstyle\sum_a \bigl|w^{\text{prev}}_a - w^{\text{target}}_a\bigr| \;\ge\; 0.10 \\
&\text{volatility regime} && \sigma_{21}/\sigma_{252} \;\ge\; 2
\end{align*}}

\block{7. Conclusions}{
\begin{itemize}
\item \textbf{The rules matter more than the model.} Four risk models within 0.04 Sharpe of each other and of $1/N$, inside bootstrap noise.
\item \textbf{CVaR is the best optimiser and the cheapest} (linear, no covariance); still not better than $1/N$.
\item \textbf{Robust is consistently worst}: guarding against the worst-case mean doubles up on rules that already limit concentration.
\item \textbf{Nothing beats the S\&P 500} in a mega-cap decade whose winners every rule-set caps. The optimisers bought drawdown, not return.
\end{itemize}
